// Employee Combo Box Items
SortByColumns(colPresenceValid, "EmployeeName", SortOrder.Ascending)

// cmbEmployee.OnChange
Set(varSelectedEmployeeKey, If(IsBlank(cmbEmployee.Selected.EmployeeKey), Blank(), cmbEmployee.Selected.EmployeeKey));
Select(btnBuildChart);

// Detail examples
// Name
LookUp(colPresenceValid, EmployeeKey = varSelectedEmployeeKey, EmployeeName)
// Manager
LookUp(colPresenceValid, EmployeeKey = varSelectedEmployeeKey, ManagerName)
// Presence
LookUp(colPresenceValid, EmployeeKey = varSelectedEmployeeKey, PresenceText)
// Direct reports
CountIf(colOrgNodes, ManagerKey = varSelectedEmployeeKey)

// Presence counters
CountIf(colPresenceValid, PresenceStatus = "In-Person")
CountIf(colPresenceValid, PresenceStatus = "Remote")
CountIf(colPresenceValid, PresenceStatus = "Vacation")
CountIf(colPresenceValid, PresenceStatus = "Sick")
CountIf(colPresenceValid, PresenceStatus = "Holiday")

// Previous weekday
Set(
    varSelectedDate,
    Switch(
        Weekday(varSelectedDate, StartOfWeek.Monday),
        1, DateAdd(varSelectedDate, -3, TimeUnit.Days),
        DateAdd(varSelectedDate, -1, TimeUnit.Days)
    )
);
Reset(dpPresenceDate);
Select(btnBuildChart);

// Next weekday
Set(
    varSelectedDate,
    Switch(
        Weekday(varSelectedDate, StartOfWeek.Monday),
        5, DateAdd(varSelectedDate, 3, TimeUnit.Days),
        DateAdd(varSelectedDate, 1, TimeUnit.Days)
    )
);
Reset(dpPresenceDate);
Select(btnBuildChart);

// Location filter OnChange (adjust selected-field name to your control)
Set(varLocationFilter, Coalesce(ddLocationFilter.Selected.Value, "All Locations"));
Select(btnBuildChart);

// Zoom initialization
Set(varZoom, 1);

// Zoom In
Set(varZoom, Min(3, varZoom * 1.2));
Select(btnBuildChart);

// Zoom Out
Set(varZoom, Max(0.5, varZoom / 1.2));
Select(btnBuildChart);

// Fit All
Set(varZoom, 1);
Set(varViewCenterX, varChartWidth / 2);
Set(varViewCenterY, varChartHeight / 2);
Select(btnBuildChart);

// Center Selection button DisplayMode
If(IsBlank(varSelectedEmployeeKey), DisplayMode.Disabled, DisplayMode.Edit)

// Center Selection OnSelect
With(
    {selectedNode: LookUp(colLayout, EmployeeKey = varSelectedEmployeeKey)},
    If(
        !IsBlank(selectedNode.EmployeeKey),
        Set(varViewCenterX, selectedNode.NodeX);
        Set(varViewCenterY, selectedNode.NodeY + varNodeHeight / 2);
        Set(varZoom, Max(varZoom, 1.5));
        Select(btnBuildChart)
    )
);

// ViewBox variables — calculate after varChartWidth/Height.
Set(varViewportWidth, varChartWidth / varZoom);
Set(varViewportHeight, varChartHeight / varZoom);
Set(varViewCenterX, If(varViewCenterX <= 0, varChartWidth / 2, varViewCenterX));
Set(varViewCenterY, If(varViewCenterY <= 0, varChartHeight / 2, varViewCenterY));
Set(varViewLeft, Max(0, Min(Max(0,varChartWidth-varViewportWidth), varViewCenterX-varViewportWidth/2)));
Set(varViewTop, Max(0, Min(Max(0,varChartHeight-varViewportHeight), varViewCenterY-varViewportHeight/2)));

// Replace fixed viewBox with:

// Data-quality examples
// Missing in-person location
Filter(colPresenceValid, WorkType = "In-Person" && IsBlank(WorkLocation))

// Self-manager
Filter(colPresenceValid, EmployeeKey = ManagerKey)

// Unplaced
Filter(
    colPresenceValid As p,
    IsBlank(LookUp(colOrgNodes, EmployeeKey = p.EmployeeKey, EmployeeKey))
)

// Diagnostics
Set(varSvgLength, Len(varOrgChartSvg));
Set(varEmployeeCount, CountRows(colPresenceValid));
Set(varLeafCount, CountRows(colLeaves));
