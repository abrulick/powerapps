// dpPresenceDate.OnChange
// Reject weekends because the source tracker only stores Monday-Friday.

If(
    Weekday(dpPresenceDate.SelectedDate, StartOfWeek.Monday) > 5,

    Notify(
        "Select a Monday-Friday date.",
        NotificationType.Warning
    );
    Reset(dpPresenceDate),

    Set(varSelectedDate, dpPresenceDate.SelectedDate);
    Set(
        varWeekStart,
        DateAdd(
            varSelectedDate,
            1 - Weekday(varSelectedDate, StartOfWeek.Monday),
            TimeUnit.Days
        )
    )
);

// Clear selected-person UI so the details never show a stale date.
Set(varSelectedEmployeeKey, Blank());
Set(varSelectedEmployeeName, Blank());
Set(varSelectedPresenceText, Blank())
