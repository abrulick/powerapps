// Set this formula on the code component's Items property.
// IMPORTANT: paste only the expression below, not "pcfOrgChart.Items =".
//
// This implementation deliberately avoids ClearCollect against SharePoint.
// The inner Filter is delegable to SharePoint and restricts the source to the
// selected week's records. AddColumns then shapes that small weekly result into
// the exact schema the PCF needs.

With(
    {
        dayNumber: Weekday(varSelectedDate, StartOfWeek.Monday),
        nextWeekDay: DateAdd(varWeekStart, 1, TimeUnit.Days)
    },
    With(
        {
            baseRows:
                AddColumns(
                    Filter(
                        'Weekly Org Tracker',
                        'Date (Mon)' >= varWeekStart &&
                        'Date (Mon)' < nextWeekDay
                    ) As wk,

                    EmployeeKey,
                        Lower(wk.'Employee Name'.Email),

                    EmployeeName,
                        wk.'Employee Name'.DisplayName,

                    ManagerKey,
                        Lower(wk.'Manager Name'.Email),

                    ManagerName,
                        wk.'Manager Name'.DisplayName,

                    WorkType,
                        Switch(
                            dayNumber,
                            1, wk.'Work Type (Mon)'.Value,
                            2, wk.'Work Type (Tue)'.Value,
                            3, wk.'Work Type (Wed)'.Value,
                            4, wk.'Work Type (Thu)'.Value,
                            5, wk.'Work Type (Fri)'.Value,
                            Blank()
                        ),

                    WorkLocation,
                        Switch(
                            dayNumber,
                            1, wk.'Location (Mon)'.Value,
                            2, wk.'Location (Tue)'.Value,
                            3, wk.'Location (Wed)'.Value,
                            4, wk.'Location (Thu)'.Value,
                            5, wk.'Location (Fri)'.Value,
                            Blank()
                        )
                )
        },
        ShowColumns(
            AddColumns(
                baseRows As employeeDay,

                PresenceStatus,
                    If(
                        employeeDay.WorkType = "In-Person" &&
                        IsBlank(employeeDay.WorkLocation),
                        "Location Missing",
                        employeeDay.WorkType
                    ),

                PresenceText,
                    Switch(
                        employeeDay.WorkType,
                        "In-Person",
                            If(
                                IsBlank(employeeDay.WorkLocation),
                                "In Person — Location Missing",
                                "In Person — " & employeeDay.WorkLocation
                            ),
                        "Remote", "Remote",
                        "Vacation", "Vacation",
                        "Sick", "Sick",
                        "Holiday", "Holiday",
                        "Unknown"
                    )
            ),
            EmployeeKey,
            EmployeeName,
            ManagerKey,
            ManagerName,
            WorkType,
            WorkLocation,
            PresenceStatus,
            PresenceText
        )
    )
)
