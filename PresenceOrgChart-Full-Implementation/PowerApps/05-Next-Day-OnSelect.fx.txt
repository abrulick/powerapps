// btnNextDay.OnSelect
// Friday jumps three calendar days to Monday; all other weekdays advance one.

Set(
    varSelectedDate,
    DateAdd(
        varSelectedDate,
        If(
            Weekday(varSelectedDate, StartOfWeek.Monday) = 5,
            3,
            1
        ),
        TimeUnit.Days
    )
);

Set(
    varWeekStart,
    DateAdd(
        varSelectedDate,
        1 - Weekday(varSelectedDate, StartOfWeek.Monday),
        TimeUnit.Days
    )
);

Reset(dpPresenceDate);

// Clear selected-person UI so the details never show a stale date.
Set(varSelectedEmployeeKey, Blank());
Set(varSelectedEmployeeName, Blank());
Set(varSelectedPresenceText, Blank())
