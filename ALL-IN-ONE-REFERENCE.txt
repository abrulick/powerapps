// ======================================================================
// PRESENCE-AWARE FULL-TREE ORG CHART
// REVIEWED + OPTIMIZED POWER FX + SVG IMPLEMENTATION
// ======================================================================
//
// IMPORTANT:
// - Source list assumed: 'Check-in'
// - Employee/Manager are SharePoint Person fields
// - Work Type and Location are single-select SharePoint Choice fields
// - This file uses US-style Power Fx separators.
// - If your authoring locale uses semicolons as argument separators,
//   translate the formula separators accordingly.
//
// The project uses TWO hidden runtime helpers:
//   btnBuildChart  = data + hierarchy + layout
//   btnRenderSvg   = SVG only
//
// This is ONE implementation architecture, not two project phases.
// ======================================================================


// ======================================================================
// SECTION 1 — scrOrgChart.OnVisible
// ======================================================================

If(
    IsBlank(varSelectedDate),

    Set(
        varSelectedDate,
        With(
            {
                wd:
                    Weekday(
                        Today(),
                        StartOfWeek.Monday
                    )
            },

            If(
                wd <= 5,
                Today(),
                DateAdd(
                    Today(),
                    5 - wd,
                    TimeUnit.Days
                )
            )
        )
    )
);

Set(varNodeWidth, 220);
Set(varNodeHeight, 92);
Set(varXGap, 44);
Set(varYGap, 76);
Set(varChartMargin, 50);

// IMPORTANT:
// Match this value to the Canvas App:
// Settings > General > Data row limit.
// 500 is the Power Apps default.
Set(varLocalRowSafetyLimit, 500);

// Use the SharePoint Choice metadata as the canonical office list.
// This keeps all configured offices available even when nobody is
// scheduled there on the selected day.
ClearCollect(
    colLocationOptions,
    {
        Value: "All Locations"
    }
);

Collect(
    colLocationOptions,
    Sort(
        Choices(
            'Check-in'.'Location (Mon)'
        ),
        Value,
        SortOrder.Ascending
    )
);

If(
    IsBlank(varLocationFilter),
    Set(varLocationFilter, "All Locations")
);

If(
    IsBlank(varZoom),
    Set(varZoom, 1)
);

If(
    IsBlank(varSelectedEmployeeKey),
    Set(varSelectedEmployeeKey, Blank())
);

Set(varResetView, true);
Set(varChartError, Blank());
Set(varChartBuilding, false);

Select(btnBuildChart);


// ======================================================================
// SECTION 2A — dpPresenceDate.DefaultDate
// ======================================================================

varSelectedDate


// ======================================================================
// SECTION 2B — dpPresenceDate.OnChange
// ======================================================================

Set(
    varSelectedDate,
    dpPresenceDate.SelectedDate
);

Set(varResetView, true);

Select(btnBuildChart);


// ======================================================================
// SECTION 3 — btnBuildChart.OnSelect
// ======================================================================

Set(varChartBuilding, true);
Set(varChartError, Blank());


// ----------------------------------------------------------------------
// 3.1 Determine selected week.
// ----------------------------------------------------------------------

Set(
    varSelectedWeekday,
    Weekday(
        varSelectedDate,
        StartOfWeek.Monday
    )
);

Set(
    varWeekStart,
    DateAdd(
        varSelectedDate,
        1 - varSelectedWeekday,
        TimeUnit.Days
    )
);


// ----------------------------------------------------------------------
// 3.2 Initialize data issue collection schema.
// ----------------------------------------------------------------------

ClearCollect(
    colDataIssues,
    {
        Severity: "",
        IssueType: "",
        EmployeeKey: "",
        EmployeeName: "",
        Message: ""
    }
);

Clear(colDataIssues);


// ----------------------------------------------------------------------
// 3.3 Reject weekend because source tracker is Monday-Friday.
// ----------------------------------------------------------------------

If(
    varSelectedWeekday > 5,

    Set(
        varChartError,
        "Select a Monday–Friday date."
    );

    Clear(colWeekRows);
    Clear(colPresence);
    Clear(colPresenceValid);
    Clear(colPresenceUnique);
    Clear(colOrgNodes);
    Clear(colLayout);

    Set(varOrgChartSvg, Blank()),


    // ==================================================================
    // WEEKDAY BUILD
    // ==================================================================


    // ------------------------------------------------------------------
    // 3.4 Load only the selected week's SharePoint rows.
    //
    // The range form is robust to a DateTime time component.
    // Filter/DateTime comparisons are delegable to SharePoint.
    // ShowColumns keeps the local copy limited to required fields.
    // The captured output is still subject to the Canvas App local
    // nondelegation row limit.
    // ------------------------------------------------------------------

    IfError(
        ClearCollect(
            colWeekRows,

            ShowColumns(
                Filter(
                    'Check-in',
                    'Date (Mon)' >= varWeekStart &&
                    'Date (Mon)' <
                        DateAdd(
                            varWeekStart,
                            1,
                            TimeUnit.Days
                        )
                ),

                ID,
                'Employee Name',
                'Manager Name',

                'Work Type (Mon)',
                'Work Type (Tue)',
                'Work Type (Wed)',
                'Work Type (Thu)',
                'Work Type (Fri)',

                'Date (Mon)',
                'Date (Tue)',
                'Date (Wed)',
                'Date (Thu)',
                'Date (Fri)',

                'Location (Mon)',
                'Location (Tue)',
                'Location (Wed)',
                'Location (Thu)',
                'Location (Fri)'
            )
        ),

        Set(
            varChartError,
            "Unable to read the SharePoint tracker: " &
            FirstError.Message
        );

        Clear(colWeekRows)
    );

    Set(
        varWeekRowCount,
        CountRows(colWeekRows)
    );

    // ClearCollect captures only up to the app's local row limit.
    // If the selected-week result reaches that limit, the chart may be
    // truncated and must not be treated as complete.
    If(
        varWeekRowCount >= varLocalRowSafetyLimit,

        Collect(
            colDataIssues,
            {
                Severity: "Warning",
                IssueType: "Local row limit risk",
                EmployeeKey: "",
                EmployeeName: "",
                Message:
                    "The selected week returned " &
                    Text(varWeekRowCount) &
                    " rows, which reached the configured safety limit of " &
                    Text(varLocalRowSafetyLimit) &
                    ". Confirm Settings > General > Data row limit and verify the week is not truncated."
            }
        )
    );


    // ------------------------------------------------------------------
    // 3.5 Empty-week handling.
    // ------------------------------------------------------------------

    If(
        !IsBlank(varChartError),

        Clear(colPresence);
        Clear(colPresenceValid);
        Clear(colPresenceUnique);
        Clear(colOrgNodes);
        Clear(colLayout);

        Set(varOrgChartSvg, Blank()),


        varWeekRowCount = 0,

        Set(
            varChartError,
            "No tracker records were found for the selected week."
        );

        Clear(colPresence);
        Clear(colPresenceValid);
        Clear(colPresenceUnique);
        Clear(colOrgNodes);
        Clear(colLayout);

        Set(varOrgChartSvg, Blank()),


        // ==============================================================
        // SELECTED WEEK HAS DATA
        // ==============================================================


        // --------------------------------------------------------------
        // 3.6 Normalize selected weekday into one common schema.
        // --------------------------------------------------------------

        ClearCollect(
            colPresence,

            ForAll(
                colWeekRows As wk,

                With(
                    {
                        selectedWorkDate:
                            Switch(
                                varSelectedWeekday,
                                1, wk.'Date (Mon)',
                                2, wk.'Date (Tue)',
                                3, wk.'Date (Wed)',
                                4, wk.'Date (Thu)',
                                5, wk.'Date (Fri)',
                                Blank()
                            ),

                        selectedWorkType:
                            Switch(
                                varSelectedWeekday,
                                1, wk.'Work Type (Mon)'.Value,
                                2, wk.'Work Type (Tue)'.Value,
                                3, wk.'Work Type (Wed)'.Value,
                                4, wk.'Work Type (Thu)'.Value,
                                5, wk.'Work Type (Fri)'.Value,
                                Blank()
                            ),

                        selectedLocation:
                            Switch(
                                varSelectedWeekday,
                                1, wk.'Location (Mon)'.Value,
                                2, wk.'Location (Tue)'.Value,
                                3, wk.'Location (Wed)'.Value,
                                4, wk.'Location (Thu)'.Value,
                                5, wk.'Location (Fri)'.Value,
                                Blank()
                            )
                    },

                    {
                        SourceID:
                            wk.ID,

                        EmployeeKey:
                            Lower(
                                Coalesce(
                                    wk.'Employee Name'.Email,
                                    ""
                                )
                            ),

                        EmployeeName:
                            Coalesce(
                                wk.'Employee Name'.DisplayName,
                                ""
                            ),

                        ManagerKey:
                            Lower(
                                Coalesce(
                                    wk.'Manager Name'.Email,
                                    ""
                                )
                            ),

                        ManagerName:
                            Coalesce(
                                wk.'Manager Name'.DisplayName,
                                ""
                            ),

                        WorkDate:
                            selectedWorkDate,

                        WorkType:
                            Coalesce(
                                selectedWorkType,
                                ""
                            ),

                        WorkLocation:
                            Coalesce(
                                selectedLocation,
                                ""
                            ),

                        PresenceStatus:
                            If(
                                selectedWorkType = "In-Person" &&
                                IsBlank(selectedLocation),

                                "Location Missing",

                                Coalesce(
                                    selectedWorkType,
                                    "Unknown"
                                )
                            ),

                        PresenceText:
                            Switch(
                                selectedWorkType,

                                "In-Person",
                                    If(
                                        IsBlank(selectedLocation),
                                        "In Person — Location Missing",
                                        "In Person — " &
                                            selectedLocation
                                    ),

                                "Remote",
                                    "Remote",

                                "Vacation",
                                    "Vacation",

                                "Sick",
                                    "Sick",

                                "Holiday",
                                    "Holiday",

                                "Unknown"
                            )
                    }
                )
            )
        );


        // --------------------------------------------------------------
        // 3.7 Flag blank employee keys.
        // --------------------------------------------------------------

        Collect(
            colDataIssues,

            ForAll(
                Filter(
                    colPresence,
                    IsBlank(EmployeeKey)
                ) As badEmployee,

                {
                    Severity: "Error",
                    IssueType: "Missing employee identity",
                    EmployeeKey: "",
                    EmployeeName: badEmployee.EmployeeName,
                    Message:
                        "Employee Name does not contain a usable email address."
                }
            )
        );


        // --------------------------------------------------------------
        // 3.8 Keep usable employee records.
        // --------------------------------------------------------------

        ClearCollect(
            colPresenceValid,

            Filter(
                colPresence,
                !IsBlank(EmployeeKey)
            )
        );


        // --------------------------------------------------------------
        // 3.9 Detect duplicate weekly employees.
        // --------------------------------------------------------------

        Collect(
            colDataIssues,

            ForAll(
                Distinct(
                    colPresenceValid,
                    EmployeeKey
                ) As employeeKeyRecord,

                With(
                    {
                        duplicateCount:
                            CountIf(
                                colPresenceValid,
                                EmployeeKey =
                                    employeeKeyRecord.Value
                            ),

                        employeeRecord:
                            First(
                                Filter(
                                    colPresenceValid,
                                    EmployeeKey =
                                        employeeKeyRecord.Value
                                )
                            )
                    },

                    If(
                        duplicateCount > 1,

                        {
                            Severity: "Error",
                            IssueType: "Duplicate employee row",
                            EmployeeKey:
                                employeeKeyRecord.Value,
                            EmployeeName:
                                employeeRecord.EmployeeName,
                            Message:
                                Text(duplicateCount) &
                                " weekly rows were found for this employee. " &
                                "The chart temporarily uses the lowest SharePoint ID row."
                        }
                    )
                )
            )
        );


        // --------------------------------------------------------------
        // 3.10 Create one deterministic visualization row per employee.
        // Lowest SharePoint SourceID wins when duplicates exist.
        // --------------------------------------------------------------

        ClearCollect(
            colPresenceUnique,

            ForAll(
                Sort(
                    Distinct(
                        colPresenceValid,
                        EmployeeKey
                    ),
                    Value,
                    SortOrder.Ascending
                ) As employeeKeyRecord,

                First(
                    SortByColumns(
                        Filter(
                            colPresenceValid,
                            EmployeeKey =
                                employeeKeyRecord.Value
                        ),
                        "SourceID",
                        SortOrder.Ascending
                    )
                )
            )
        );


        // --------------------------------------------------------------
        // 3.11 Additional source-quality checks.
        // --------------------------------------------------------------

        Collect(
            colDataIssues,

            ForAll(
                Filter(
                    colPresenceUnique,
                    EmployeeKey = ManagerKey &&
                    !IsBlank(EmployeeKey)
                ) As selfManager,

                {
                    Severity: "Error",
                    IssueType: "Self manager",
                    EmployeeKey: selfManager.EmployeeKey,
                    EmployeeName: selfManager.EmployeeName,
                    Message:
                        "Employee and manager email are the same."
                }
            )
        );

        Collect(
            colDataIssues,

            ForAll(
                Filter(
                    colPresenceUnique,
                    IsBlank(WorkType)
                ) As missingWorkType,

                {
                    Severity: "Warning",
                    IssueType: "Missing Work Type",
                    EmployeeKey: missingWorkType.EmployeeKey,
                    EmployeeName: missingWorkType.EmployeeName,
                    Message:
                        "No Work Type is recorded for the selected date."
                }
            )
        );

        Collect(
            colDataIssues,

            ForAll(
                Filter(
                    colPresenceUnique,

                    !IsBlank(WorkType) &&

                    !(
                        WorkType in
                        [
                            "In-Person",
                            "Remote",
                            "Vacation",
                            "Sick",
                            "Holiday"
                        ]
                    )
                ) As invalidWorkType,

                {
                    Severity: "Warning",
                    IssueType: "Unexpected Work Type",
                    EmployeeKey: invalidWorkType.EmployeeKey,
                    EmployeeName: invalidWorkType.EmployeeName,
                    Message:
                        "Unexpected Work Type: " &
                        invalidWorkType.WorkType
                }
            )
        );

        Collect(
            colDataIssues,

            ForAll(
                Filter(
                    colPresenceUnique,
                    WorkType = "In-Person" &&
                    IsBlank(WorkLocation)
                ) As missingLocation,

                {
                    Severity: "Warning",
                    IssueType: "Missing Location",
                    EmployeeKey: missingLocation.EmployeeKey,
                    EmployeeName: missingLocation.EmployeeName,
                    Message:
                        "Employee is In-Person but no office location is recorded."
                }
            )
        );

        Collect(
            colDataIssues,

            ForAll(
                Filter(
                    colPresenceUnique,
                    WorkType <> "In-Person" &&
                    !IsBlank(WorkLocation)
                ) As staleLocation,

                {
                    Severity: "Warning",
                    IssueType: "Location on non-In-Person day",
                    EmployeeKey: staleLocation.EmployeeKey,
                    EmployeeName: staleLocation.EmployeeName,
                    Message:
                        "A location is stored even though Work Type is " &
                        staleLocation.WorkType &
                        ". The chart ignores this location."
                }
            )
        );

        Collect(
            colDataIssues,

            ForAll(
                Filter(
                    colPresenceUnique,
                    !IsBlank(WorkDate) &&
                    WorkDate <> varSelectedDate
                ) As wrongDate,

                {
                    Severity: "Warning",
                    IssueType: "Date mismatch",
                    EmployeeKey: wrongDate.EmployeeKey,
                    EmployeeName: wrongDate.EmployeeName,
                    Message:
                        "The selected weekday Date column contains " &
                        Text(
                            wrongDate.WorkDate,
                            "[$-en-US]m/d/yyyy"
                        ) &
                        " instead of " &
                        Text(
                            varSelectedDate,
                            "[$-en-US]m/d/yyyy"
                        ) &
                        "."
                }
            )
        );


        // --------------------------------------------------------------
        // 3.12 If no usable employees remain, stop.
        // --------------------------------------------------------------

        If(
            CountRows(colPresenceUnique) = 0,

            Set(
                varChartError,
                "The selected week contains no usable employee records."
            );

            Clear(colOrgNodes);
            Clear(colLayout);

            Set(varOrgChartSvg, Blank()),


            // ==========================================================
            // BUILD HIERARCHY
            // ==========================================================


            // ----------------------------------------------------------
            // 3.13 LEVEL 0 — local root(s)
            // ----------------------------------------------------------

            ClearCollect(
                colLevel0,

                ForAll(
                    SortByColumns(
                        Filter(
                            colPresenceUnique As employee,

                            IsBlank(employee.ManagerKey) ||

                            IsBlank(
                                LookUp(
                                    colPresenceUnique,
                                    EmployeeKey =
                                        employee.ManagerKey,
                                    EmployeeKey
                                )
                            )
                        ),
                        "EmployeeName",
                        SortOrder.Ascending
                    ) As root,

                    {
                        SourceID: root.SourceID,
                        EmployeeKey: root.EmployeeKey,
                        EmployeeName: root.EmployeeName,
                        ManagerKey: root.ManagerKey,
                        ManagerName: root.ManagerName,
                        WorkDate: root.WorkDate,
                        WorkType: root.WorkType,
                        WorkLocation: root.WorkLocation,
                        PresenceStatus: root.PresenceStatus,
                        PresenceText: root.PresenceText,

                        Level: 0,

                        PathKey:
                            "/" &
                            root.EmployeeKey &
                            "/",

                        SortPath:
                            "/" &
                            Lower(root.EmployeeName) &
                            "|" &
                            root.EmployeeKey &
                            "/"
                    }
                )
            );


            // ----------------------------------------------------------
            // 3.13A Validate local roots.
            // ----------------------------------------------------------

            If(
                CountRows(colLevel0) = 0,

                Set(
                    varChartError,
                    "No local hierarchy root could be identified. Check manager cycles or self-manager records."
                )
            );

            If(
                CountRows(colLevel0) > 1,

                Collect(
                    colDataIssues,
                    {
                        Severity: "Warning",
                        IssueType: "Multiple local roots",
                        EmployeeKey: "",
                        EmployeeName: "",
                        Message:
                            Text(
                                CountRows(colLevel0)
                            ) &
                            " local roots were found. This may be valid, or it may indicate missing manager rows in the selected week."
                    }
                )
            );


            // ----------------------------------------------------------
            // 3.14 LEVEL 1
            // ----------------------------------------------------------

            ClearCollect(
                colLevel1,

                ForAll(
                    SortByColumns(
                        Filter(
                            colPresenceUnique As child,

                            !IsBlank(
                                LookUp(
                                    colLevel0,
                                    EmployeeKey =
                                        child.ManagerKey,
                                    EmployeeKey
                                )
                            )
                        ),
                        "EmployeeName",
                        SortOrder.Ascending
                    ) As child,

                    With(
                        {
                            parent:
                                LookUp(
                                    colLevel0,
                                    EmployeeKey =
                                        child.ManagerKey
                                )
                        },

                        {
                            SourceID: child.SourceID,
                            EmployeeKey: child.EmployeeKey,
                            EmployeeName: child.EmployeeName,
                            ManagerKey: child.ManagerKey,
                            ManagerName: child.ManagerName,
                            WorkDate: child.WorkDate,
                            WorkType: child.WorkType,
                            WorkLocation: child.WorkLocation,
                            PresenceStatus: child.PresenceStatus,
                            PresenceText: child.PresenceText,

                            Level: 1,

                            PathKey:
                                parent.PathKey &
                                child.EmployeeKey &
                                "/",

                            SortPath:
                                parent.SortPath &
                                Lower(child.EmployeeName) &
                                "|" &
                                child.EmployeeKey &
                                "/"
                        }
                    )
                )
            );


            // ----------------------------------------------------------
            // 3.15 LEVEL 2
            // ----------------------------------------------------------

            ClearCollect(
                colLevel2,

                ForAll(
                    SortByColumns(
                        Filter(
                            colPresenceUnique As child,

                            !IsBlank(
                                LookUp(
                                    colLevel1,
                                    EmployeeKey =
                                        child.ManagerKey,
                                    EmployeeKey
                                )
                            )
                        ),
                        "EmployeeName",
                        SortOrder.Ascending
                    ) As child,

                    With(
                        {
                            parent:
                                LookUp(
                                    colLevel1,
                                    EmployeeKey =
                                        child.ManagerKey
                                )
                        },

                        {
                            SourceID: child.SourceID,
                            EmployeeKey: child.EmployeeKey,
                            EmployeeName: child.EmployeeName,
                            ManagerKey: child.ManagerKey,
                            ManagerName: child.ManagerName,
                            WorkDate: child.WorkDate,
                            WorkType: child.WorkType,
                            WorkLocation: child.WorkLocation,
                            PresenceStatus: child.PresenceStatus,
                            PresenceText: child.PresenceText,
                            Level: 2,
                            PathKey:
                                parent.PathKey &
                                child.EmployeeKey &
                                "/",
                            SortPath:
                                parent.SortPath &
                                Lower(child.EmployeeName) &
                                "|" &
                                child.EmployeeKey &
                                "/"
                        }
                    )
                )
            );


            // ----------------------------------------------------------
            // 3.16 LEVEL 3
            // ----------------------------------------------------------

            ClearCollect(
                colLevel3,

                ForAll(
                    SortByColumns(
                        Filter(
                            colPresenceUnique As child,

                            !IsBlank(
                                LookUp(
                                    colLevel2,
                                    EmployeeKey =
                                        child.ManagerKey,
                                    EmployeeKey
                                )
                            )
                        ),
                        "EmployeeName",
                        SortOrder.Ascending
                    ) As child,

                    With(
                        {
                            parent:
                                LookUp(
                                    colLevel2,
                                    EmployeeKey =
                                        child.ManagerKey
                                )
                        },

                        {
                            SourceID: child.SourceID,
                            EmployeeKey: child.EmployeeKey,
                            EmployeeName: child.EmployeeName,
                            ManagerKey: child.ManagerKey,
                            ManagerName: child.ManagerName,
                            WorkDate: child.WorkDate,
                            WorkType: child.WorkType,
                            WorkLocation: child.WorkLocation,
                            PresenceStatus: child.PresenceStatus,
                            PresenceText: child.PresenceText,
                            Level: 3,
                            PathKey:
                                parent.PathKey &
                                child.EmployeeKey &
                                "/",
                            SortPath:
                                parent.SortPath &
                                Lower(child.EmployeeName) &
                                "|" &
                                child.EmployeeKey &
                                "/"
                        }
                    )
                )
            );


            // ----------------------------------------------------------
            // 3.17 LEVEL 4
            // ----------------------------------------------------------

            ClearCollect(
                colLevel4,

                ForAll(
                    SortByColumns(
                        Filter(
                            colPresenceUnique As child,

                            !IsBlank(
                                LookUp(
                                    colLevel3,
                                    EmployeeKey =
                                        child.ManagerKey,
                                    EmployeeKey
                                )
                            )
                        ),
                        "EmployeeName",
                        SortOrder.Ascending
                    ) As child,

                    With(
                        {
                            parent:
                                LookUp(
                                    colLevel3,
                                    EmployeeKey =
                                        child.ManagerKey
                                )
                        },

                        {
                            SourceID: child.SourceID,
                            EmployeeKey: child.EmployeeKey,
                            EmployeeName: child.EmployeeName,
                            ManagerKey: child.ManagerKey,
                            ManagerName: child.ManagerName,
                            WorkDate: child.WorkDate,
                            WorkType: child.WorkType,
                            WorkLocation: child.WorkLocation,
                            PresenceStatus: child.PresenceStatus,
                            PresenceText: child.PresenceText,
                            Level: 4,
                            PathKey:
                                parent.PathKey &
                                child.EmployeeKey &
                                "/",
                            SortPath:
                                parent.SortPath &
                                Lower(child.EmployeeName) &
                                "|" &
                                child.EmployeeKey &
                                "/"
                        }
                    )
                )
            );


            // ----------------------------------------------------------
            // 3.18 LEVEL 5
            // ----------------------------------------------------------

            ClearCollect(
                colLevel5,

                ForAll(
                    SortByColumns(
                        Filter(
                            colPresenceUnique As child,

                            !IsBlank(
                                LookUp(
                                    colLevel4,
                                    EmployeeKey =
                                        child.ManagerKey,
                                    EmployeeKey
                                )
                            )
                        ),
                        "EmployeeName",
                        SortOrder.Ascending
                    ) As child,

                    With(
                        {
                            parent:
                                LookUp(
                                    colLevel4,
                                    EmployeeKey =
                                        child.ManagerKey
                                )
                        },

                        {
                            SourceID: child.SourceID,
                            EmployeeKey: child.EmployeeKey,
                            EmployeeName: child.EmployeeName,
                            ManagerKey: child.ManagerKey,
                            ManagerName: child.ManagerName,
                            WorkDate: child.WorkDate,
                            WorkType: child.WorkType,
                            WorkLocation: child.WorkLocation,
                            PresenceStatus: child.PresenceStatus,
                            PresenceText: child.PresenceText,
                            Level: 5,
                            PathKey:
                                parent.PathKey &
                                child.EmployeeKey &
                                "/",
                            SortPath:
                                parent.SortPath &
                                Lower(child.EmployeeName) &
                                "|" &
                                child.EmployeeKey &
                                "/"
                        }
                    )
                )
            );


            // ----------------------------------------------------------
            // 3.19 LEVEL 6
            // ----------------------------------------------------------

            ClearCollect(
                colLevel6,

                ForAll(
                    SortByColumns(
                        Filter(
                            colPresenceUnique As child,

                            !IsBlank(
                                LookUp(
                                    colLevel5,
                                    EmployeeKey =
                                        child.ManagerKey,
                                    EmployeeKey
                                )
                            )
                        ),
                        "EmployeeName",
                        SortOrder.Ascending
                    ) As child,

                    With(
                        {
                            parent:
                                LookUp(
                                    colLevel5,
                                    EmployeeKey =
                                        child.ManagerKey
                                )
                        },

                        {
                            SourceID: child.SourceID,
                            EmployeeKey: child.EmployeeKey,
                            EmployeeName: child.EmployeeName,
                            ManagerKey: child.ManagerKey,
                            ManagerName: child.ManagerName,
                            WorkDate: child.WorkDate,
                            WorkType: child.WorkType,
                            WorkLocation: child.WorkLocation,
                            PresenceStatus: child.PresenceStatus,
                            PresenceText: child.PresenceText,
                            Level: 6,
                            PathKey:
                                parent.PathKey &
                                child.EmployeeKey &
                                "/",
                            SortPath:
                                parent.SortPath &
                                Lower(child.EmployeeName) &
                                "|" &
                                child.EmployeeKey &
                                "/"
                        }
                    )
                )
            );


            // ----------------------------------------------------------
            // 3.20 LEVEL 7
            // ----------------------------------------------------------

            ClearCollect(
                colLevel7,

                ForAll(
                    SortByColumns(
                        Filter(
                            colPresenceUnique As child,

                            !IsBlank(
                                LookUp(
                                    colLevel6,
                                    EmployeeKey =
                                        child.ManagerKey,
                                    EmployeeKey
                                )
                            )
                        ),
                        "EmployeeName",
                        SortOrder.Ascending
                    ) As child,

                    With(
                        {
                            parent:
                                LookUp(
                                    colLevel6,
                                    EmployeeKey =
                                        child.ManagerKey
                                )
                        },

                        {
                            SourceID: child.SourceID,
                            EmployeeKey: child.EmployeeKey,
                            EmployeeName: child.EmployeeName,
                            ManagerKey: child.ManagerKey,
                            ManagerName: child.ManagerName,
                            WorkDate: child.WorkDate,
                            WorkType: child.WorkType,
                            WorkLocation: child.WorkLocation,
                            PresenceStatus: child.PresenceStatus,
                            PresenceText: child.PresenceText,
                            Level: 7,
                            PathKey:
                                parent.PathKey &
                                child.EmployeeKey &
                                "/",
                            SortPath:
                                parent.SortPath &
                                Lower(child.EmployeeName) &
                                "|" &
                                child.EmployeeKey &
                                "/"
                        }
                    )
                )
            );


            // ----------------------------------------------------------
            // 3.21 Combine all hierarchy levels.
            // ----------------------------------------------------------

            ClearCollect(
                colOrgNodes,
                colLevel0
            );

            Collect(
                colOrgNodes,
                colLevel1,
                colLevel2,
                colLevel3,
                colLevel4,
                colLevel5,
                colLevel6,
                colLevel7
            );


            // ----------------------------------------------------------
            // 3.22 Find unplaced employees.
            // ----------------------------------------------------------

            ClearCollect(
                colUnplaced,

                Filter(
                    colPresenceUnique As person,

                    IsBlank(
                        LookUp(
                            colOrgNodes,
                            EmployeeKey =
                                person.EmployeeKey,
                            EmployeeKey
                        )
                    )
                )
            );

            Collect(
                colDataIssues,

                ForAll(
                    colUnplaced As unplacedPerson,

                    {
                        Severity: "Error",
                        IssueType: "Unplaced hierarchy node",
                        EmployeeKey: unplacedPerson.EmployeeKey,
                        EmployeeName: unplacedPerson.EmployeeName,
                        Message:
                            "Employee could not be placed in hierarchy levels 0–7. " &
                            "Check manager relationships, cycles, self-manager records, or hierarchy depth."
                    }
                )
            );


            // ----------------------------------------------------------
            // 3.23 Identify and sort leaf nodes.
            // ----------------------------------------------------------

            ClearCollect(
                colLeavesBase,

                SortByColumns(
                    Filter(
                        colOrgNodes As employee,

                        CountIf(
                            colOrgNodes,
                            ManagerKey =
                                employee.EmployeeKey
                        ) = 0
                    ),
                    "SortPath",
                    SortOrder.Ascending
                )
            );


            // ----------------------------------------------------------
            // 3.24 Assign deterministic X position to every leaf.
            // ----------------------------------------------------------

            ClearCollect(
                colLeaves,

                ForAll(
                    Sequence(
                        CountRows(colLeavesBase)
                    ) As sequenceRecord,

                    With(
                        {
                            leaf:
                                Index(
                                    colLeavesBase,
                                    sequenceRecord.Value
                                )
                        },

                        {
                            SourceID: leaf.SourceID,
                            EmployeeKey: leaf.EmployeeKey,
                            EmployeeName: leaf.EmployeeName,
                            ManagerKey: leaf.ManagerKey,
                            ManagerName: leaf.ManagerName,
                            WorkDate: leaf.WorkDate,
                            WorkType: leaf.WorkType,
                            WorkLocation: leaf.WorkLocation,
                            PresenceStatus: leaf.PresenceStatus,
                            PresenceText: leaf.PresenceText,
                            Level: leaf.Level,
                            PathKey: leaf.PathKey,
                            SortPath: leaf.SortPath,

                            LeafIndex:
                                sequenceRecord.Value,

                            LeafX:
                                varChartMargin +
                                varNodeWidth / 2 +
                                (
                                    sequenceRecord.Value - 1
                                ) *
                                (
                                    varNodeWidth +
                                    varXGap
                                )
                        }
                    )
                )
            );


            // ----------------------------------------------------------
            // 3.25 Calculate every node's X/Y coordinates.
            //
            // NodeX = average X position of all descendant leaves.
            // ----------------------------------------------------------

            ClearCollect(
                colLayout,

                AddColumns(
                    colOrgNodes As node,

                    DirectReportCount,
                        CountIf(
                            colOrgNodes,
                            ManagerKey =
                                node.EmployeeKey
                        ),

                    NodeX,
                        Average(
                            Filter(
                                colLeaves As leaf,

                                StartsWith(
                                    leaf.PathKey,
                                    node.PathKey
                                )
                            ),
                            LeafX
                        ),

                    NodeY,
                        varChartMargin +
                        node.Level *
                        (
                            varNodeHeight +
                            varYGap
                        )
                )
            );


            If(
                CountRows(colLayout) = 0,

                Set(
                    varChartError,
                    Coalesce(
                        varChartError,
                        "The organization hierarchy could not be laid out. Review the data issues list."
                    )
                )
            );


            // ----------------------------------------------------------
            // 3.26 Calculate chart dimensions.
            // ----------------------------------------------------------

            Set(
                varChartWidth,

                Max(
                    800,

                    Max(
                        colLayout,
                        NodeX
                    ) +
                    varNodeWidth / 2 +
                    varChartMargin
                )
            );

            Set(
                varChartHeight,

                Max(
                    500,

                    Max(
                        colLayout,
                        NodeY
                    ) +
                    varNodeHeight +
                    varChartMargin
                )
            );


            // ----------------------------------------------------------
            // 3.27 Validate current location filter.
            //
            // colLocationOptions is loaded once from the canonical
            // SharePoint Location (Mon) Choice metadata in OnVisible.
            // ----------------------------------------------------------

            If(
                IsBlank(
                    LookUp(
                        colLocationOptions,
                        Value =
                            varLocationFilter,
                        Value
                    )
                ),

                Set(
                    varLocationFilter,
                    "All Locations"
                );

                Reset(ddLocationFilter)
            );


            // ----------------------------------------------------------
            // 3.28 Clear employee selection if employee isn't present.
            // ----------------------------------------------------------

            If(
                !IsBlank(varSelectedEmployeeKey) &&

                IsBlank(
                    LookUp(
                        colPresenceUnique,
                        EmployeeKey =
                            varSelectedEmployeeKey,
                        EmployeeKey
                    )
                ),

                Set(
                    varSelectedEmployeeKey,
                    Blank()
                );

                Reset(cmbEmployee)
            );


            // ----------------------------------------------------------
            // 3.29 Initialize/reset viewport after a data/date rebuild.
            // ----------------------------------------------------------

            If(
                varResetView ||
                IsBlank(varViewCenterX) ||
                IsBlank(varViewCenterY),

                Set(varZoom, 1);
                Set(
                    varViewCenterX,
                    varChartWidth / 2
                );
                Set(
                    varViewCenterY,
                    varChartHeight / 2
                )
            );

            Set(varResetView, false);


            // ----------------------------------------------------------
            // 3.30 Render the chart.
            // ----------------------------------------------------------

            Select(btnRenderSvg)
        )
    )
);

Set(varLastBuildTime, Now());
Set(varChartBuilding, false);
