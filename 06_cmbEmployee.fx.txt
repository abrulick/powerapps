// ======================================================================
// SECTION 6A — cmbEmployee.Items
// ======================================================================

SortByColumns(
    colPresenceUnique,
    "EmployeeName",
    SortOrder.Ascending
)

// ======================================================================
// SECTION 6B — cmbEmployee.DefaultSelectedItems
// ======================================================================

If(
    IsBlank(varSelectedEmployeeKey),
    Filter(
        colPresenceUnique,
        false
    ),

    Filter(
        colPresenceUnique,
        EmployeeKey =
            varSelectedEmployeeKey
    )
)

// ======================================================================
// SECTION 6C — cmbEmployee.OnChange
// ======================================================================

Set(
    varSelectedEmployeeKey,

    If(
        IsBlank(
            cmbEmployee.Selected.EmployeeKey
        ),
        Blank(),
        cmbEmployee.Selected.EmployeeKey
    )
);

Select(btnRenderSvg);
