// ======================================================================
// SECTION 14A — btnRefreshData.OnSelect
// ======================================================================

Set(varChartBuilding, true);

IfError(
    Refresh(
        'Weekly Org Tracker'
    );

    Set(varLastSourceRefresh, Now());
    Set(varResetView, true);

    Select(btnBuildChart),

    Set(varChartBuilding, false);

    Set(
        varChartError,
        "Unable to refresh the SharePoint tracker: " &
        FirstError.Message
    )
);

// ======================================================================
// SECTION 14B — lblLastRefresh.Text
// ======================================================================

If(
    IsBlank(varLastSourceRefresh),

    "Source refresh: app load",

    "Source refreshed: " &
    Text(
        varLastSourceRefresh,
        "[$-en-US]m/d/yyyy h:mm AM/PM"
    )
)
